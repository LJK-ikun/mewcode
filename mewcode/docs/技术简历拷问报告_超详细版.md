### 🔍 Q1: "只传名称不传完整schema"是如何实现的？

#### 源码位置

**文件**: `mewcode/tools/__init__.py`
**行数**: 116-134

#### 完整源码（逐行注释）

```python
def search_deferred(self, query: str) -> list[dict[str, str]]:
    """
    搜索延迟加载的工具（只返回名称+描述）
  
    参数:
        query: 用户搜索的关键词（如"team", "memory"）
  
    返回:
        工具列表，每个工具只包含name和description，不包含完整schema
    """
    # 初始化结果列表
    results = []
  
    # 遍历所有延迟加载的工具名称
    # _deferred_tool_names 是一个set，存储了所有延迟工具的名称
    for tool_name in self._deferred_tool_names:
        # 获取延迟工具对象（此时才真正加载）
        # _get_deferred_tool() 会动态导入工具模块
        tool = self._get_deferred_tool(tool_name)
    
        # 如果工具存在 且 查询关键词在工具名称中
        if tool and query.lower() in tool.name.lower():
            # 关键：只添加name和description
            # 不包含params_schema（参数定义）
            results.append({
                "name": tool.name,              # 工具名称（如"Agent"）
                "description": tool.description, # 工具描述（一句话说明）
                # 注意：没有 "params_schema" 字段
                # 完整schema包含的字段有：
                # - type: "object"
                # - properties: {...}  (所有参数定义)
                # - required: [...]     (必需参数列表)
                # 这些都没有包含，所以Token占用大幅减少
            })
  
    # 返回简化的工具列表
    return results
```

#### 延迟工具的注册过程（逐行注释）

```python
def register_deferred(self, tool_name: str) -> None:
    """
    注册一个延迟加载的工具
  
    关键：这里只记录工具名称，不加载Tool对象
  
    参数:
        tool_name: 工具名称（如"Agent", "TeamCreate"）
    """
    # 只将工具名称添加到集合中
    # 不调用任何导入或实例化操作
    # 不占用内存，不生成schema
    self._deferred_tool_names.add(tool_name)
  
    # 对比：非延迟注册的做法
    # self._tools[tool_name] = tool_instance  # 立即实例化，占用内存
    # schema = tool_instance.get_schema()      # 立即生成完整schema


def mark_discovered(self, tool_name: str) -> None:
    """
    标记工具已被发现，触发完整加载
  
    调用时机：LLM调用ToolSearch后
  
    参数:
        tool_name: 被发现的工具名称
    """
    # 检查是否在延迟工具列表中
    if tool_name not in self._deferred_tool_names:
        return  # 不是延迟工具，无需处理
  
    # 真正加载工具（动态导入）
    tool = self._get_deferred_tool(tool_name)
  
    # 如果加载成功
    if tool:
        # 将工具从延迟列表移除
        self._deferred_tool_names.remove(tool_name)
    
        # 添加到正式的工具注册表
        self._tools[tool.name] = tool
    
        # 现在这个工具的完整schema会被发送给LLM
        # LLM可以看到所有参数定义并调用这个工具


def _get_deferred_tool(self, tool_name: str) -> Tool | None:
    """
    动态加载延迟工具
  
    这个方法执行实际的导入和实例化
  
    参数:
        tool_name: 工具名称
  
    返回:
        Tool对象，如果加载失败则返回None
    """
    try:
        # 根据工具名称构建模块路径
        # 例如：tool_name="Agent" → module_path="mewcode.tools.impl.agent_tool"
        module_path = f"mewcode.tools.impl.{tool_name.lower()}_tool"
    
        # 动态导入模块
        # importlib.import_module() 只在此时才加载Python模块
        module = importlib.import_module(module_path)
    
        # 获取Tool类（约定：类名=工具名+"Tool"）
        # 例如：AgentTool, TeamCreateTool
        tool_class = getattr(module, f"{tool_name}Tool")
    
        # 实例化Tool对象
        # 传入必要的依赖（registry, agent等）
        tool_instance = tool_class(
            registry=self,
            agent=self._agent,
            # ... 其他依赖
        )
    
        # 返回完整的Tool对象
        return tool_instance
    
    except (ImportError, AttributeError) as e:
        # 如果导入失败，记录日志并返回None
        log.warning(f"Failed to load deferred tool {tool_name}: {e}")
        return None
```

### 🔍 Q2: "LLM按需发现"的机制是什么？

#### 源码位置

**类**: `ToolSearchTool`

#### 完整源码（逐行注释）

```python
from pydantic import BaseModel, Field

from mewcode.tools.base import Tool, ToolResult


class ToolSearchParams(BaseModel):
    """
    ToolSearch工具的参数定义
    """
    # 搜索关键词（LLM提供）
    query: str = Field(
        description="Keyword to search for tools (e.g., 'team', 'memory', 'git')"
    )


class ToolSearchTool(Tool):
    """
    工具搜索工具
  
    职责：
    1. 接收LLM的搜索请求
    2. 在延迟工具中搜索匹配的工具
    3. 触发匹配工具的完整加载
    4. 返回搜索结果给LLM
  
    关键：这是一个普通工具，LLM主动调用
    """
  
    # 工具名称（LLM通过这个名称调用）
    name = "ToolSearch"
  
    # 工具描述（LLM看到这个描述决定是否调用）
    description = "Search for additional tools by keyword. Use when you need tools not currently available."
  
    # 参数schema（自动从ToolSearchParams生成）
    params_schema = ToolSearchParams
  
  
    def __init__(self, registry: "ToolRegistry", **kwargs):
        """
        初始化
    
        参数:
            registry: 工具注册表（用于搜索延迟工具）
        """
        super().__init__(**kwargs)
        # 保存注册表引用（用于调用search_deferred()）
        self.registry = registry
  
  
    async def execute(self, params: BaseModel) -> ToolResult:
        """
        执行工具搜索
    
        这个方法在LLM调用ToolSearch时被执行
    
        参数:
            params: ToolSearchParams对象，包含query字段
    
        返回:
            ToolResult对象，包含搜索结果文本
        """
        # 1. 类型转换（BaseModel → ToolSearchParams）
        p: ToolSearchParams = params
    
        # 2. 调用注册表的搜索方法
        # registry.search_deferred() 在所有延迟工具中搜索
        # 返回：[{"name": "Agent", "description": "..."}, ...]
        results = self.registry.search_deferred(p.query)
    
        # 3. 如果没有找到匹配的工具
        if not results:
            # 返回"未找到"的消息给LLM
            return ToolResult(
                output=f"No tools found matching '{p.query}'"
            )
    
        # 4. 标记所有找到的工具为"已发现"
        # 这会触发完整schema的加载
        for r in results:
            # mark_discovered() 会：
            # - 动态导入工具模块
            # - 实例化Tool对象
            # - 将工具添加到_tools字典
            # - 下一轮LLM调用时，这个工具的完整schema会被发送
            self.registry.mark_discovered(r["name"])
    
        # 5. 构建返回消息
        # 告诉LLM找到了哪些工具
        tool_list = "\n".join([
            f"- {r['name']}: {r['description']}"
            for r in results
        ])
    
        output = (
            f"Found {len(results)} tool(s) matching '{p.query}':\n\n"
            f"{tool_list}\n\n"
            f"These tools are now available for use."
        )
    
        # 6. 返回结果
        # LLM会看到这个消息，知道现在可以使用这些工具了
        return ToolResult(output=output)
```

def build_tool_search_reminder() -> str:
    """
    构建工具搜索提醒
  
    这段文本会被插入到系统提示中
    引导LLM在需要时主动调用ToolSearch
  
    返回:
        提醒文本
    """
    return """
## Additional Tools

If you need tools not currently in your toolset, use ToolSearch to discover them.

Available tool categories:
- **team**: TeamCreate, SendMessage, TeamList (for multi-agent collaboration)
- **memory**: RememberTool, ForgetTool (for persistent memory)
- **git**: GitCommit, GitPush (for version control operations)
- **workflow**: Workflow (for multi-agent orchestration)

Example:
```

To create a team, first search for team tools:
ToolSearch(query="team")

```

After searching, the tools will become available in your toolset.
"""
    # 关键：
    # 1. 明确告诉LLM有哪些工具类别
    # 2. 给出使用示例
    # 3. 说明搜索后工具会变为可用


# 系统提示的完整构建过程
def build_system_prompt(agent: Agent) -> str:
    """
    构建完整的系统提示
  
    参数:
        agent: Agent对象
  
    返回:
        完整的系统提示文本
    """
    # 基础系统提示
    prompt = BASE_SYSTEM_PROMPT
  
    # 添加工具列表（当前可用的工具）
    prompt += "\n\n## Available Tools\n\n"
    for tool in agent.registry.list_tools():
        # 只列出已加载的工具
        # 延迟工具不会出现在这里
        prompt += f"- {tool.name}: {tool.description}\n"
  
    # 添加工具搜索提醒
    if agent.registry.has_deferred_tools():
        # 只有存在延迟工具时才添加提醒
        prompt += "\n" + build_tool_search_reminder()
  
    return prompt
```

# === 可以运行的测量脚本 ===

import json

def count_tokens_approximate(text: str) -> int:
    """
    估算Token数量
  
    规则（基于Claude tokenizer的近似）：
    - 英文单词：约1.3字符/token
    - JSON标点：约5字符/token
    - 综合：约4.5字符/token
  
    参数:
        text: 要统计的文本
  
    返回:
        估算的token数
    """
    # 使用字符数除以4.5作为近似
    return int(len(text) / 4.5)


def measure_single_tool_reduction(tool_name: str) -> dict:
    """
    测量单个工具的Token减少比例
  
    参数:
        tool_name: 工具名称（如"Agent"）
  
    返回:
        {
            "tool": 工具名称,
            "full_tokens": 完整schema的token数,
            "deferred_tokens": 延迟schema的token数,
            "reduction": 减少比例,
            "saved_tokens": 节省的token数
        }
    """
    # 1. 获取工具对象
    from mewcode.tools import ToolRegistry
    registry = ToolRegistry()
    tool = registry.get_tool(tool_name)
  
    # 2. 获取完整schema
    full_schema = tool.get_schema()
    full_json = json.dumps(full_schema, indent=2)
    full_tokens = count_tokens_approximate(full_json)
  
    # 3. 构建延迟schema
    deferred_schema = {
        "name": tool.name,
        "description": tool.description
    }
    deferred_json = json.dumps(deferred_schema, indent=2)
    deferred_tokens = count_tokens_approximate(deferred_json)
  
    # 4. 计算减少比例
    saved_tokens = full_tokens - deferred_tokens
    reduction = saved_tokens / full_tokens if full_tokens > 0 else 0
  
    return {
        "tool": tool_name,
        "full_tokens": full_tokens,
        "deferred_tokens": deferred_tokens,
        "reduction": reduction,
        "saved_tokens": saved_tokens
    }


def measure_all_tools() -> dict:
    """
    测量所有工具的平均Token减少比例
  
    返回:
        {
            "total_tools": 工具总数,
            "full_tokens": 完整模式总token数,
            "deferred_tokens": 延迟模式总token数,
            "total_reduction": 总减少比例,
            "average_reduction": 平均减少比例,
            "per_tool_stats": 每个工具的详细统计
        }
    """
    from mewcode.tools import ToolRegistry
  
    # 1. 初始化注册表
    registry = ToolRegistry()
  
    # 2. 获取所有工具
    all_tools = registry.list_tools()
  
    # 3. 测量每个工具
    per_tool_stats = []
    total_full = 0
    total_deferred = 0
  
    for tool in all_tools:
        # 测量单个工具
        stats = measure_single_tool_reduction(tool.name)
        per_tool_stats.append(stats)
    
        # 累加总数
        total_full += stats["full_tokens"]
        total_deferred += stats["deferred_tokens"]
  
    # 4. 计算总体统计
    total_saved = total_full - total_deferred
    total_reduction = total_saved / total_full if total_full > 0 else 0
  
    # 5. 计算平均减少比例
    reductions = [s["reduction"] for s in per_tool_stats]
    average_reduction = sum(reductions) / len(reductions) if reductions else 0
  
    return {
        "total_tools": len(all_tools),
        "full_tokens": total_full,
        "deferred_tokens": total_deferred,
        "total_saved": total_saved,
        "total_reduction": total_reduction,
        "average_reduction": average_reduction,
        "per_tool_stats": per_tool_stats
    }


# === 使用示例 ===

if __name__ == "__main__":
    # 测量所有工具
    result = measure_all_tools()

    print(f"完整模式总Token: {result['full_tokens']}")
    print(f"延迟模式总Token: {result['deferred_tokens']}")
    print(f"节省Token: {result['total_saved']}")
    print(f"总减少比例: {result['total_reduction']:.1%}")
    print(f"平均减少比例: {result['average_reduction']:.1%}")
  
    # 输出每个工具的详细统计

              f"完整: {stat['full_tokens']:4d} tokens | "

### 🔍 Q1: "智能替换+滑动窗口双层策略"的具体实现是什么？

#### 源码位置

**文件**: `mewcode/conversation/compressor.py`
**类**: `ConversationCompressor`

#### 完整源码（逐行注释）

```python
from typing import List, Dict, Any
from dataclasses import dataclass

from mewcode.llm import LLMClient
from mewcode.conversation.message import Message, Role


@dataclass
class CompressionWindow:
    """
    压缩窗口配置
  
    定义哪些消息保留全文，哪些需要压缩
    """
    # 保留最近N条消息的全文（不压缩）
    keep_recent: int = 20
  
    # 历史消息的摘要比例（压缩后的大小 / 原始大小）
    summary_ratio: float = 0.3
  
    # 触发压缩的阈值（当前token数 / 最大token数）
    compression_threshold: float = 0.8


class ConversationCompressor:
    """
    对话压缩器
  
    职责：
    1. 监控对话的token使用情况
    2. 当接近上下文限制时触发压缩
    3. 使用双层策略保留重要信息
  
    双层策略：
    - 第一层（智能替换）：识别重要消息，用摘要替换不重要的
    - 第二层（滑动窗口）：保留最近N条消息全文，压缩更早的消息
    """
  
    def __init__(
        self,
        llm_client: LLMClient,
        max_tokens: int = 200_000,
        window: CompressionWindow = None
    ):
        """
        初始化压缩器
    
        参数:
            llm_client: LLM客户端（用于生成摘要）
            max_tokens: 上下文窗口大小（默认200K for Claude Sonnet 4）
            window: 压缩窗口配置
        """
        # 保存LLM客户端引用（用于调用summarize）
        self.llm_client = llm_client
    
        # 保存最大token限制
        self.max_tokens = max_tokens
    
        # 使用默认配置或用户提供的配置
        self.window = window or CompressionWindow()
    
        # 压缩统计
        self.compression_count = 0  # 压缩次数
        self.total_saved_tokens = 0  # 累计节省的tokens
  
  
    def should_compress(self, messages: List[Message]) -> bool:
        """
        判断是否需要压缩
    
        参数:
            messages: 当前的消息列表
    
        返回:
            True表示需要压缩，False表示不需要
        """
        # 1. 计算当前消息的总token数
        current_tokens = sum(msg.count_tokens() for msg in messages)
    
        # 2. 计算使用率
        usage_ratio = current_tokens / self.max_tokens
    
        # 3. 判断是否超过阈值
        # 默认阈值是0.8，即使用了80%的上下文窗口时触发压缩
        should_compress = usage_ratio >= self.window.compression_threshold
    
        # 4. 记录日志
        if should_compress:
            log.info(
                f"Compression triggered: {current_tokens}/{self.max_tokens} "
                f"({usage_ratio:.1%}) >= {self.window.compression_threshold:.1%}"
            )
    
        return should_compress
  
  
    async def compress(self, messages: List[Message]) -> List[Message]:
        """
        执行压缩
    
        这是核心方法，实现双层压缩策略
    
        参数:
            messages: 原始消息列表
    
        返回:
            压缩后的消息列表
        """
        # === 第一层：滑动窗口分割 ===
    
        # 1. 保留最近的N条消息（全文）
        # 这些消息通常是当前任务相关的，必须保留完整信息
        recent_messages = messages[-self.window.keep_recent:]
    
        # 2. 剩余的是历史消息（需要压缩）
        # 这些消息可能包含重要的决策或上下文，但不需要全文
        historical_messages = messages[:-self.window.keep_recent]
    
        # 3. 如果没有历史消息，无需压缩
        if not historical_messages:
            return messages
    
        # === 第二层：智能替换 ===
    
        # 4. 对历史消息进行重要性分类
        # 这个方法会分析每条消息的内容，识别：
        # - 重要决策（保留全文）
        # - 代码修改记录（保留全文）
        # - 工具调用结果（可压缩）
        # - 普通对话（可压缩）
        classified = self._classify_messages(historical_messages)
    
        # 5. 提取需要保留全文的重要消息
        important_full = classified["important_full"]
    
        # 6. 提取可压缩的消息
        compressible = classified["compressible"]
    
        # 7. 对可压缩的消息生成摘要
        # 这里调用LLM生成智能摘要，而不是简单截断
        summarized = await self._summarize_messages(compressible)
    
        # 8. 组合压缩后的消息列表
        compressed_messages = (
            important_full +      # 重要历史消息（全文）
            [summarized] +        # 可压缩消息的摘要（单条合并消息）
            recent_messages       # 最近消息（全文）
        )
    
        # 9. 统计压缩效果
        original_tokens = sum(msg.count_tokens() for msg in messages)
        compressed_tokens = sum(msg.count_tokens() for msg in compressed_messages)
        saved_tokens = original_tokens - compressed_tokens
    
        self.compression_count += 1
        self.total_saved_tokens += saved_tokens
    
        log.info(
            f"Compression #{self.compression_count}: "
            f"{original_tokens} → {compressed_tokens} tokens "
            f"(saved {saved_tokens}, {saved_tokens/original_tokens:.1%})"
        )
    
        # 10. 返回压缩后的消息列表
        return compressed_messages
  
  
    def _classify_messages(
        self,
        messages: List[Message]
    ) -> Dict[str, List[Message]]:
        """
        对消息进行重要性分类
    
        这是智能替换策略的核心：不是简单地压缩所有历史消息，
        而是识别哪些消息包含重要信息，必须保留全文
    
        参数:
            messages: 待分类的消息列表
    
        返回:
            {
                "important_full": 需要保留全文的重要消息,
                "compressible": 可以压缩的消息
            }
        """
        # 初始化分类结果
        important_full = []
        compressible = []
    
        # 遍历每条消息
        for msg in messages:
            # === 重要性判断规则 ===
        
            # 规则1：用户的明确决策
            # 例如："不要使用X库，改用Y库"
            if self._is_user_decision(msg):
                important_full.append(msg)
                continue
        
            # 规则2：代码修改记录
            # 包含Edit/Write工具调用的消息
            if self._is_code_change(msg):
                important_full.append(msg)
                continue
        
            # 规则3：错误和修复
            # 包含错误信息和对应修复的消息
            if self._is_error_fix(msg):
                important_full.append(msg)
                continue
        
            # 规则4：配置和环境信息
            # 例如：项目结构、依赖版本
            if self._is_config_info(msg):
                important_full.append(msg)
                continue
        
            # 规则5：其他消息可压缩
            # 例如：文件读取结果、Bash输出、普通对话
            compressible.append(msg)
    
        return {
            "important_full": important_full,
            "compressible": compressible
        }
  
  
    def _is_user_decision(self, msg: Message) -> bool:
        """
        判断是否是用户的明确决策
    
        特征：
        - Role是USER
        - 包含明确的指示词："不要"、"改用"、"必须"、"禁止"
        - 或者是对AI提议的明确回应："对"、"不对"、"是的"、"错了"
    
        参数:
            msg: 消息对象
    
        返回:
            True表示是决策消息
        """
        # 只检查用户消息
        if msg.role != Role.USER:
            return False
    
        # 获取消息文本
        text = msg.get_text_content()
    
        # 定义决策关键词
        decision_keywords = [
            "不要", "不用", "改用", "换成",
            "必须", "一定要", "务必",
            "禁止", "不能", "别",
            "对", "不对", "是的", "不是",
            "错了", "正确", "就这样"
        ]
    
        # 检查是否包含决策关键词
        return any(keyword in text for keyword in decision_keywords)
  
  
    def _is_code_change(self, msg: Message) -> bool:
        """
        判断是否是代码修改记录
    
        特征：
        - Role是ASSISTANT
        - 包含Edit或Write工具调用
        - 或者文本中明确说明了修改内容
    
        参数:
            msg: 消息对象
    
        返回:
            True表示是代码修改
        """
        # 只检查助手消息
        if msg.role != Role.ASSISTANT:
            return False
    
        # 检查是否包含工具调用
        tool_calls = msg.get_tool_calls()
    
        # 如果有Edit或Write工具调用，这是代码修改
        for tool_call in tool_calls:
            if tool_call.name in ["Edit", "Write"]:
                return True
    
        # 检查文本是否描述了修改
        text = msg.get_text_content()
        change_keywords = ["修改了", "添加了", "删除了", "更新了", "创建了"]
    
        return any(keyword in text for keyword in change_keywords)
  
  
    def _is_error_fix(self, msg: Message) -> bool:
        """
        判断是否是错误和修复记录
    
        特征：
        - 包含错误信息（Error、Exception、Failed）
        - 或者紧跟在错误消息后的修复说明
    
        参数:
            msg: 消息对象
    
        返回:
            True表示是错误相关
        """
        # 获取消息内容
        content = msg.get_all_content()
    
        # 定义错误关键词
        error_keywords = [
            "Error", "Exception", "Failed", "错误",
            "失败", "异常", "报错", "修复"
        ]
    
        # 检查内容是否包含错误关键词
        return any(keyword in content for keyword in error_keywords)
  
  
    def _is_config_info(self, msg: Message) -> bool:
        """
        判断是否是配置和环境信息
    
        特征：
        - 包含package.json、requirements.txt等配置文件
        - 包含项目结构信息
        - 包含环境变量或系统信息
    
        参数:
            msg: 消息对象
    
        返回:
            True表示是配置信息
        """
        content = msg.get_all_content()
    
        # 定义配置关键词
        config_keywords = [
            "package.json", "requirements.txt", "Cargo.toml",
            "pom.xml", "build.gradle", ".env",
            "项目结构", "依赖", "版本"
        ]
    
        return any(keyword in content for keyword in config_keywords)
  
  
    async def _summarize_messages(
        self,
        messages: List[Message]
    ) -> Message:
        """
        生成消息摘要
    
        这里使用LLM生成智能摘要，而不是简单的截断或删除
        关键：摘要保留核心信息，但用更少的tokens
    
        参数:
            messages: 待摘要的消息列表
    
        返回:
            包含摘要的单条消息
        """
        # 1. 如果消息列表为空，返回空摘要
        if not messages:
            return Message(
                role=Role.ASSISTANT,
                content="[Empty summary]"
            )
    
        # 2. 构建摘要提示
        # 这个提示告诉LLM如何生成高质量的摘要
        summary_prompt = """
请为以下对话生成简洁摘要。保留：
1. 核心技术决策和原因
2. 重要的代码修改（文件名和主要变更）
3. 遇到的关键问题和解决方案
4. 用户的明确指示

忽略：
1. 文件读取的详细内容
2. Bash命令的详细输出
3. 重复的讨论

摘要应比原文短70%左右，但保留所有关键信息。

原始对话：
---
"""
    
        # 3. 添加原始消息内容
        for msg in messages:
            summary_prompt += f"\n[{msg.role.value}]\n{msg.get_all_content()}\n"
    
        summary_prompt += "\n---\n请生成摘要："
    
        # 4. 调用LLM生成摘要
        # 使用较小的模型（如Haiku）以降低成本
        summary_response = await self.llm_client.generate(
            prompt=summary_prompt,
            model="claude-haiku-4.5",  # 使用快速模型
            max_tokens=4000  # 限制摘要长度
        )
    
        # 5. 提取摘要文本
        summary_text = summary_response.content[0].text
    
        # 6. 添加压缩标记
        # 这样用户可以看到这是压缩后的内容
        marked_summary = (
            f"[📦 压缩摘要 - 原始{len(messages)}条消息]\n\n"
            f"{summary_text}\n\n"
            f"[压缩比例: {self.window.summary_ratio:.0%}]"
        )
    
        # 7. 创建摘要消息
        summary_message = Message(
            role=Role.ASSISTANT,
            content=marked_summary,
            metadata={
                "is_summary": True,
                "original_count": len(messages),
                "compression_ratio": self.window.summary_ratio
            }
        )
    
        # 8. 验证压缩效果
        original_tokens = sum(msg.count_tokens() for msg in messages)
        summary_tokens = summary_message.count_tokens()
        actual_ratio = summary_tokens / original_tokens
    
        # 如果实际压缩比例不理想，记录警告
        if actual_ratio > self.window.summary_ratio * 1.5:
            log.warning(
                f"Summary compression ratio {actual_ratio:.1%} "
                f"exceeds target {self.window.summary_ratio:.1%}"
            )
    
        # 9. 返回摘要消息
        return summary_message
```

```python
import asyncio
import time
from typing import List

from mewcode.conversation import ConversationManager
from mewcode.conversation.message import Message, Role
from mewcode.conversation.compressor import ConversationCompressor


class LongSessionTest:
    """
    长会话测试
  
    目标：验证压缩后的信息保留质量
    方法：模拟10小时会话，测试AI能否引用前期决策
    """
  
    def __init__(self):
        self.manager = ConversationManager()
        self.compressor = ConversationCompressor(
            llm_client=self.manager.llm_client
        )
    
        # 记录关键决策的位置
        self.decision_positions = []
  
  
    async def simulate_10_hour_session(self) -> dict:
        """
        模拟10小时会话
    
        假设：
        - 每轮对话1分钟（用户输入 + AI响应 + 工具调用）
        - 10小时 = 600轮对话
        - 在不同时间点插入关键决策
    
        返回:
            测试结果统计
        """
        print("开始模拟10小时会话...")
        start_time = time.time()
    
        # 1. 创建600轮对话
        for round_num in range(600):
            # 每100轮插入一个关键决策
            if round_num % 100 == 15:
                # 插入决策消息
                decision = self._create_decision_message(round_num)
                self.manager.add_message(decision)
            
                # 记录决策位置和内容
                self.decision_positions.append({
                    "round": round_num,
                    "message_index": len(self.manager.messages) - 1,
                    "content": decision.content,
                    "keyword": f"DECISION_{round_num}"  # 唯一标识
                })
        
            else:
                # 插入普通对话
                user_msg = Message(
                    role=Role.USER,
                    content=f"Round {round_num}: user request"
                )
            
                assistant_msg = Message(
                    role=Role.ASSISTANT,
                    content=f"Round {round_num}: assistant response",
                    tool_calls=[self._create_random_tool_call()]
                )
            
                self.manager.add_message(user_msg)
                self.manager.add_message(assistant_msg)
        
            # 每80轮触发一次压缩（模拟达到80%阈值）
            if round_num % 80 == 79:
                print(f"  Round {round_num}: 触发压缩...")
                await self._compress_and_verify()
    
        elapsed = time.time() - start_time
        print(f"模拟完成，耗时: {elapsed:.1f}秒")
    
        # 2. 最终压缩
        print("执行最终压缩...")
        await self._compress_and_verify()
    
        # 3. 测试引用能力
        print("测试前期决策引用...")
        results = await self._test_decision_recall()
    
        return results
  
  
    def _create_decision_message(self, round_num: int) -> Message:
        """
        创建关键决策消息
    
        每个决策都有唯一标识，方便后续验证
    
        参数:
            round_num: 轮次编号
    
        返回:
            决策消息
        """
        # 创建有实际意义的决策内容
        decisions = [
            "不要使用Redis缓存，改用本地内存，因为是单实例部署",
            "API超时设置为30秒，不要改成更长，因为前端有60秒总超时限制",
            "日志级别设置为INFO，不要改成DEBUG，因为会产生太多日志",
            "数据库连接池大小设置为5，不要改大，因为RDS只支持10个连接",
            "不要使用Pandas，改用Polars，因为需要更好的性能",
            "不要使用SQLAlchemy，改用原生SQL，因为查询逻辑太复杂"
        ]
    
        decision_text = decisions[round_num // 100 % len(decisions)]
    
        return Message(
            role=Role.USER,
            content=f"[DECISION_{round_num}] {decision_text}"
        )
  
  
    def _create_random_tool_call(self) -> dict:
        """
        创建随机工具调用（模拟正常工作流程）
        """
        import random
    
        tools = ["Read", "Write", "Edit", "Bash", "Grep"]
        tool = random.choice(tools)
    
        return {
            "name": tool,
            "input": {"param": f"value_{random.randint(1000, 9999)}"}
        }
  
  
    async def _compress_and_verify(self):
        """
        执行压缩并验证关键决策是否保留
        """
        # 执行压缩
        compressed = await self.compressor.compress(self.manager.messages)
    
        # 更新消息列表
        self.manager.messages = compressed
    
        # 验证所有决策是否保留
        for decision in self.decision_positions:
            keyword = decision["keyword"]
        
            # 检查压缩后的消息中是否包含这个决策
            found = any(
                keyword in msg.get_all_content()
                for msg in compressed
            )
        
            if not found:
                print(f"  ⚠️  决策 {keyword} 未找到！")
            else:
                print(f"  ✅ 决策 {keyword} 保留")
  
  
    async def _test_decision_recall(self) -> dict:
        """
        测试AI能否准确引用前期决策
    
        方法：
        1. 对每个早期决策，构造一个相关问题
        2. 让AI回答，看是否能引用决策
        3. 统计准确率
    
        返回:
            {
                "total_decisions": 总决策数,
                "correctly_recalled": 正确引用数,
                "accuracy": 准确率,
                "examples": 示例
            }
        """
        from mewcode.llm import LLMClient
    
        llm = LLMClient()
    
        total = len(self.decision_positions)
        correctly_recalled = 0
        examples = []
    
        # 只测试前3个决策（早期的，最容易丢失）
        for decision_info in self.decision_positions[:3]:
            keyword = decision_info["keyword"]
            content = decision_info["content"]
        
            # 根据决策内容构造问题
            question = self._generate_question_for_decision(content)
        
            # 将问题添加到对话中
            self.manager.add_message(Message(
                role=Role.USER,
                content=question
            ))
        
            # 让AI回答
            response = await llm.generate(
                messages=self.manager.messages,
                model="claude-sonnet-5"
            )
        
            response_text = response.content[0].text
        
            # 检查回答是否包含决策关键词
            # 或者提到了决策中的关键信息
            if keyword in response_text or self._check_decision_mentioned(content, response_text):
                correctly_recalled += 1
                result = "✅ 正确引用"
            else:
                result = "❌ 未能引用"
        
            examples.append({
                "decision": content,
                "question": question,
                "response": response_text[:200],  # 前200字符
                "result": result
            })
        
            print(f"  {result}: {keyword}")
    
        accuracy = correctly_recalled / total if total > 0 else 0
    
        return {
            "total_decisions": total,
            "correctly_recalled": correctly_recalled,
            "accuracy": accuracy,
            "examples": examples
        }
  
  
    def _generate_question_for_decision(self, decision_content: str) -> str:
        """
        根据决策内容生成相关问题
    
        例如：
        - 决策："不要使用Redis，改用本地内存"
        - 问题："为什么我们不使用Redis做缓存？"
        """
        if "Redis" in decision_content:
            return "为什么我们不使用Redis做缓存？有什么替代方案吗？"
        elif "超时" in decision_content:
            return "API超时时间应该设置为多少？为什么？"
        elif "日志级别" in decision_content:
            return "生产环境的日志级别应该是什么？为什么不用DEBUG？"
        elif "连接池" in decision_content:
            return "数据库连接池大小是多少？可以调大吗？"
        elif "Pandas" in decision_content:
            return "为什么我们用Polars而不是Pandas？"
        elif "SQLAlchemy" in decision_content:
            return "为什么不使用SQLAlchemy ORM？"
        else:
            return "请回顾一下我们之前的技术决策"
  
  
    def _check_decision_mentioned(self, decision: str, response: str) -> bool:
        """
        检查AI的回答是否提到了决策中的关键信息
    
        例如：
        - 决策："不要使用Redis，因为单实例部署"
        - 回答包含："单实例"或"本地内存" → True
        """
        # 提取决策中的关键词
        keywords = []
    
        if "Redis" in decision:
            keywords.extend(["本地内存", "单实例"])
        if "超时" in decision:
            keywords.extend(["30秒", "前端", "60秒"])
        if "日志" in decision:
            keywords.extend(["INFO", "日志量"])
        if "连接池" in decision:
            keywords.extend(["5", "RDS", "10个连接"])
        if "Pandas" in decision:
            keywords.extend(["Polars", "性能"])
        if "SQLAlchemy" in decision:
            keywords.extend(["原生SQL", "复杂"])
    
        # 检查回答是否包含至少2个关键词
        matched = sum(1 for keyword in keywords if keyword in response)
    
        return matched >= 2

from enum import Enum


class HookEvent(Enum):
    """
    Hook事件枚举
  
    定义了MewCode支持的所有生命周期事件
    每个事件都在特定时机触发，允许用户介入决策流程
    """
  
    # === 会话生命周期事件 (3个) ===
  
    SESSION_START = "session_start"
    """
    会话启动事件
  
    触发时机：Agent初始化完成，开始接收用户输入之前
    使用场景：
    - 加载项目配置
    - 初始化环境变量
    - 打印欢迎信息
    - 检查系统依赖
  
    Payload:
    {
        "agent_id": "agent_abc123",
        "working_dir": "/path/to/project",
        "config": {...}
    }
    """
  
    SESSION_END = "session_end"
    """
    会话结束事件
  
    触发时机：用户退出或会话异常终止时
    使用场景：
    - 清理临时文件
    - 保存会话日志
    - 上传统计数据
    - 关闭数据库连接
  
    Payload:
    {
        "agent_id": "agent_abc123",
        "duration_seconds": 3600,
        "message_count": 150,
        "exit_reason": "user_quit" | "error" | "timeout"
    }
    """
  
    MESSAGE_RECEIVED = "message_received"
    """
    消息接收事件
  
    触发时机：收到用户输入后，AI处理之前
    使用场景：
    - 内容过滤（敏感词检测）
    - 消息预处理（格式化、翻译）
    - 审计日志（记录用户输入）
    - 权限检查（某些命令需要授权）
  
    Payload:
    {
        "message": {
            "role": "user",
            "content": "用户输入的文本"
        },
        "agent_id": "agent_abc123"
    }
  
    可拦截：返回 {"allow": false} 可阻止消息处理
    """
  
    # === 工具调用生命周期事件 (4个) ===
  
    TOOL_CALL_BEFORE = "tool_call_before"
    """
    工具调用前事件
  
    触发时机：AI决定调用工具，但尚未执行
    使用场景：
    - 权限检查（限制某些工具的使用）
    - 参数校验（防止危险操作）
    - 成本控制（限制昂贵的API调用）

    - 结果过滤（隐藏敏感信息）
    - 结果增强（添加额外上下文）

    Payload:
    {
        "tool_name": "Bash",
        "tool_input": {"command": "ls -la"},
        "tool_output": "total 48\ndrwxr-xr-x ...",
        "duration_ms": 150,
        "agent_id": "agent_abc123"

    """

    触发时机：工具执行失败时
    使用场景：
    - 错误处理（自动重试）
    - 错误日志（记录详细堆栈）
    - 告警通知（关键操作失败告警）
    - 降级处理（使用备用方案）
  
    Payload:
    {
        "tool_name": "Read",

        "error": "FileNotFoundError: ...",

    """
  
    FILE_WRITE_BEFORE = "file_write_before"
    """
    文件写入前事件
  
    触发时机：Write/Edit工具写入文件之前
    使用场景：
    - 写入保护（防止覆盖重要文件）
    - 格式化（自动格式化代码）
    - 备份（写入前自动备份）

    - 参数校验（检查SubAgent配置）
  
    Payload:

        "prompt": "SubAgent的任务描述",
        "config": {...}
    }
  
    可拦截：返回 {"allow": false} 可阻止创建
    """
  
    AGENT_SPAWN_AFTER = "agent_spawn_after"

```python
import subprocess
import asyncio
import json
from typing import Any, Dict
from abc import ABC, abstractmethod

import httpx

from mewcode.llm import LLMClient
from mewcode.agent import Agent


class HookExecutor(ABC):
    """
    Hook执行器抽象基类
  
    定义了所有执行器的通用接口
    """
  
    @abstractmethod
    async def execute(self, event: str, payload: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行Hook
    
        参数:
            event: 事件名称（如"tool_call_before"）
            payload: 事件数据
    
        返回:
            执行结果，可能包含：
            - allow: bool（是否允许继续）
            - modified_*: 修改后的数据
            - reason: 拦截原因
        """
        pass


class CommandExecutor(HookExecutor):
    """
    命令执行器
  
    功能：执行shell命令作为Hook
    使用场景：
    - 运行linter检查代码
    - 执行测试
    - 调用外部脚本
    - 触发构建
  
    配置示例：
    {
        "event": "file_write_before",
        "condition": "path =~ \\.py$",
        "executor": {
            "type": "command",
            "command": "black {file_path}",  # 自动格式化
            "timeout": 30
        }
    }
    """
  
    def __init__(self, command: str, timeout: int = 30):
        """
        初始化命令执行器
    
        参数:
            command: 要执行的shell命令（可包含占位符）
            timeout: 超时时间（秒）
        """
        # 保存命令模板
        # 模板中可以使用占位符，如{file_path}、{tool_name}
        self.command_template = command
    
        # 超时时间
        self.timeout = timeout
  
  
    async def execute(self, event: str, payload: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行命令
    
        流程：
        1. 替换命令模板中的占位符
        2. 执行shell命令
        3. 解析命令输出，决定是否允许操作
    
        参数:
            event: 事件名称
            payload: 事件数据
    
        返回:
            {
                "allow": true | false,
                "reason": "命令输出" | "命令失败原因",
                "command_output": "...",
                "exit_code": 0 | 1
            }
        """
        # 1. 替换占位符
        # 例如：
        # 模板："black {file_path}"
        # payload: {"file_path": "src/main.py"}
        # 结果："black src/main.py"
        command = self.command_template.format(**payload)
    
        try:
            # 2. 执行命令
            # asyncio.create_subprocess_shell() 异步执行shell命令
            process = await asyncio.create_subprocess_shell(
                command,
                stdout=asyncio.subprocess.PIPE,  # 捕获标准输出
                stderr=asyncio.subprocess.PIPE,  # 捕获标准错误
                shell=True  # 使用shell解释器
            )
        
            # 3. 等待命令完成（带超时）
            try:
                stdout, stderr = await asyncio.wait_for(
                    process.communicate(),  # 等待进程结束并获取输出
                    timeout=self.timeout
                )
            except asyncio.TimeoutError:
                # 超时：杀死进程
                process.kill()
                await process.wait()
            
                return {
                    "allow": False,
                    "reason": f"命令执行超时（>{self.timeout}秒）",
                    "command": command
                }
        
            # 4. 解码输出
            stdout_text = stdout.decode("utf-8", errors="ignore")
            stderr_text = stderr.decode("utf-8", errors="ignore")
        
            # 5. 根据退出码决定是否允许
            # 约定：退出码0表示允许，非0表示拒绝
            if process.returncode == 0:
                return {
                    "allow": True,
                    "command_output": stdout_text,
                    "exit_code": 0
                }
            else:
                return {
                    "allow": False,
                    "reason": f"命令失败：{stderr_text}",
                    "command_output": stdout_text,
                    "exit_code": process.returncode
                }
    
        except Exception as e:
            # 6. 异常处理
            return {
                "allow": False,
                "reason": f"命令执行异常：{str(e)}",
                "command": command
            }


class PromptExecutor(HookExecutor):
    """
    提示词执行器
  
    功能：使用LLM分析事件并做决策
    使用场景：
    - 智能代码审查（AI判断代码质量）
    - 智能权限控制（AI判断操作是否合理）
    - 内容审核（AI检测敏感内容）
    - 智能建议（AI提供优化建议）
  
    配置示例：
    {
        "event": "tool_call_before",
        "condition": "tool_name == 'Bash' and 'rm' in tool_input.command",
        "executor": {
            "type": "prompt",
            "prompt": "分析这个命令是否安全：{tool_input.command}",
            "model": "claude-sonnet-5"
        }
    }
    """
  
    def __init__(
        self,
        prompt_template: str,
        llm_client: LLMClient,
        model: str = "claude-sonnet-5"
    ):
        """
        初始化提示词执行器
    
        参数:
            prompt_template: 提示词模板（可包含占位符）
            llm_client: LLM客户端
            model: 使用的模型
        """
        # 保存提示词模板
        self.prompt_template = prompt_template
    
        # 保存LLM客户端引用
        self.llm_client = llm_client
    
        # 使用的模型
        self.model = model
  
  
    async def execute(self, event: str, payload: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行AI决策
    
        流程：
        1. 构建完整提示词
        2. 调用LLM分析
        3. 解析LLM响应，提取决策
    
        参数:
            event: 事件名称
            payload: 事件数据
    
        返回:
            {
                "allow": true | false,
                "reason": "AI的分析理由",
                "suggestions": ["AI的建议1", "AI的建议2"]
            }
        """
        # 1. 替换提示词模板中的占位符
        # 使用json.dumps()将复杂对象转换为可读字符串
        formatted_payload = {
            k: json.dumps(v, ensure_ascii=False) if isinstance(v, (dict, list)) else v
            for k, v in payload.items()
        }
    
        prompt = self.prompt_template.format(**formatted_payload)
    
        # 2. 构建完整的分析提示
        # 添加系统指令，确保AI按特定格式输出
        full_prompt = f"""
你是一个代码安全和质量分析专家。请分析以下事件并给出决策。

事件类型：{event}
事件数据：
{json.dumps(payload, ensure_ascii=False, indent=2)}

分析任务：
{prompt}

请以JSON格式输出你的决策：
{{
    "allow": true或false,
    "reason": "你的分析理由",
    "suggestions": ["建议1", "建议2"]
}}

要求：
1. 如果操作安全且合理，设置allow=true
2. 如果操作有风险或不合理，设置allow=false
3. reason必须清晰说明判断依据
4. suggestions可以为空数组
5. 只输出JSON，不要额外文本
"""
    
        try:
            # 3. 调用LLM
            response = await self.llm_client.generate(
                prompt=full_prompt,
                model=self.model,
                max_tokens=1000  # 限制输出长度
            )
        
            # 4. 提取响应文本
            response_text = response.content[0].text.strip()
        
            # 5. 解析JSON
            # 尝试提取JSON块（可能包含在```json...```中）
            if "```json" in response_text:
                # 提取```json和```之间的内容
                json_start = response_text.find("```json") + 7
                json_end = response_text.find("```", json_start)
                json_text = response_text[json_start:json_end].strip()
            elif "```" in response_text:
                # 提取```和```之间的内容
                json_start = response_text.find("```") + 3
                json_end = response_text.find("```", json_start)
                json_text = response_text[json_start:json_end].strip()
            else:
                # 直接使用全文
                json_text = response_text
        
            # 6. 解析为字典
            result = json.loads(json_text)
        
            # 7. 验证必需字段
            if "allow" not in result:
                result["allow"] = False
                result["reason"] = "AI未给出明确决策"
        
            return result
    
        except json.JSONDecodeError as e:
            # JSON解析失败
            return {
                "allow": False,
                "reason": f"AI响应格式错误：{str(e)}",
                "raw_response": response_text
            }
    
        except Exception as e:
            # 其他异常
            return {
                "allow": False,
                "reason": f"AI分析失败：{str(e)}"
            }


class HttpExecutor(HookExecutor):
    """
    HTTP执行器
  
    功能：通过HTTP请求对接外部系统
    使用场景：
    - 对接审批系统（需要人工审批）
    - 对接合规检查平台
    - 发送通知（Slack、钉钉）
    - 记录审计日志到外部系统
  
    配置示例：
    {
        "event": "git_push_before",
        "condition": "branch == 'main'",
        "executor": {
            "type": "http",
            "url": "https://approval.company.com/api/check",
            "method": "POST",
            "headers": {
                "Authorization": "Bearer ${APPROVAL_TOKEN}"
            },
            "timeout": 60
        }
    }
    """
  
    def __init__(
        self,
        url: str,
        method: str = "POST",
        headers: Dict[str, str] = None,
        timeout: int = 30
    ):
        """
        初始化HTTP执行器
    
        参数:
            url: 目标URL
            method: HTTP方法（GET/POST/PUT）
            headers: HTTP头（可包含环境变量引用）
            timeout: 超时时间（秒）
        """
        # 保存URL
        self.url = url
    
        # HTTP方法（转为大写）
        self.method = method.upper()
    
        # 处理headers中的环境变量
        # 例如："Bearer ${TOKEN}" → "Bearer actual_token_value"
        self.headers = self._resolve_env_vars(headers or {})
    
        # 超时时间
        self.timeout = timeout
  
  
    def _resolve_env_vars(self, headers: Dict[str, str]) -> Dict[str, str]:
        """
        解析headers中的环境变量引用
    
        格式：${VAR_NAME}
    
        参数:
            headers: 原始headers
    
        返回:
            解析后的headers
        """
        import os
        import re
    
        resolved = {}
    
        # 正则表达式：匹配${...}格式
        env_var_pattern = re.compile(r'\$\{([A-Z_]+)\}')
    
        for key, value in headers.items():
            # 查找所有${VAR}
            matches = env_var_pattern.findall(value)
        
            # 替换每个匹配项
            resolved_value = value
            for var_name in matches:
                # 从环境变量获取值
                env_value = os.getenv(var_name, "")
            
                # 替换
                resolved_value = resolved_value.replace(f"${{{var_name}}}", env_value)
        
            resolved[key] = resolved_value
    
        return resolved
  
  
    async def execute(self, event: str, payload: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行HTTP请求
    
        流程：
        1. 构建请求体（包含event和payload）
        2. 发送HTTP请求
        3. 解析响应，提取决策
    
        参数:
            event: 事件名称
            payload: 事件数据
    
        返回:
            外部系统的响应（应包含allow字段）
        """
        # 1. 构建请求体
        request_body = {
            "event": event,
            "payload": payload,
            "timestamp": self._get_timestamp()
        }
    
        try:
            # 2. 创建HTTP客户端
            async with httpx.AsyncClient(timeout=self.timeout) as client:
                # 3. 发送请求
                if self.method == "GET":
                    # GET请求：将payload作为查询参数
                    response = await client.get(
                        self.url,
                        params={"event": event, **payload},
                        headers=self.headers
                    )
                else:
                    # POST/PUT请求：将payload作为JSON body
                    response = await client.request(
                        method=self.method,
                        url=self.url,
                        json=request_body,
                        headers=self.headers
                    )
            
                # 4. 检查HTTP状态码
                # 2xx表示成功
                if 200 <= response.status_code < 300:
                    # 5. 解析响应JSON
                    result = response.json()
                
                    # 6. 确保包含allow字段
                    if "allow" not in result:
                        # 默认：HTTP成功 = 允许操作
                        result["allow"] = True
                
                    return result
            
                else:
                    # HTTP错误
                    return {
                        "allow": False,
                        "reason": f"HTTP {response.status_code}: {response.text}",
                        "status_code": response.status_code
                    }
    
        except httpx.TimeoutException:
            # 超时
            return {
                "allow": False,
                "reason": f"HTTP请求超时（>{self.timeout}秒）",
                "url": self.url
            }
    
        except Exception as e:
            # 其他异常
            return {
                "allow": False,
                "reason": f"HTTP请求失败：{str(e)}",
                "url": self.url
            }
  
  
    def _get_timestamp(self) -> str:
        """
        获取当前时间戳（ISO 8601格式）
    
        返回:
            时间戳字符串（如"2024-01-15T10:30:00Z"）
        """
        from datetime import datetime, timezone
    
        return datetime.now(timezone.utc).isoformat()


class AgentExecutor(HookExecutor):
    """
    Agent执行器
  
    功能：创建SubAgent执行复杂的Hook逻辑
    使用场景：
    - 复杂的代码审查（需要读取多个文件）
    - 自动修复（发现问题后自动修复）
    - 智能决策（需要多步推理）
    - 报告生成（生成详细的分析报告）
  
    配置示例：
    {
        "event": "git_commit_before",
        "condition": "len(files) > 10",
        "executor": {
            "type": "agent",
            "prompt": "检查这些文件的修改是否符合代码规范：{files}",
            "model": "claude-sonnet-5"
        }
    }
    """
  
    def __init__(
        self,
        prompt_template: str,
        agent_factory,  # 用于创建Agent的工厂函数
        model: str = "claude-sonnet-5"
    ):
        """
        初始化Agent执行器
    
        参数:
            prompt_template: 提示词模板
            agent_factory: Agent工厂（用于创建SubAgent）
            model: 使用的模型
        """
        # 保存提示词模板
        self.prompt_template = prompt_template
    
        # 保存Agent工厂
        self.agent_factory = agent_factory
    
        # 使用的模型
        self.model = model
  
  
    async def execute(self, event: str, payload: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行Agent分析
    
        流程：
        1. 创建SubAgent
        2. 将事件数据传递给SubAgent
        3. 等待SubAgent完成
        4. 提取SubAgent的结论
    
        参数:
            event: 事件名称
            payload: 事件数据
    
        返回:
            SubAgent的分析结果
        """
        # 1. 替换提示词模板
        formatted_payload = {
            k: json.dumps(v, ensure_ascii=False) if isinstance(v, (dict, list)) else v
            for k, v in payload.items()
        }
    
        prompt = self.prompt_template.format(**formatted_payload)
    
        # 2. 构建完整的分析任务
        full_prompt = f"""
你是一个代码分析Agent，正在处理一个Hook事件。

事件类型：{event}
事件数据：
{json.dumps(payload, ensure_ascii=False, indent=2)}

任务：
{prompt}

要求：
1. 仔细分析事件数据
2. 如果需要，可以使用Read工具读取相关文件
3. 给出明确的allow决策（true或false）
4. 提供详细的reason说明
5. 如果有建议，在suggestions中列出

最后，请以JSON格式输出：
{{
    "allow": true或false,
    "reason": "详细分析理由",
    "suggestions": ["建议1", "建议2"]
}}
"""
    
        try:
            # 3. 创建SubAgent
            # agent_factory可能是Agent类或创建Agent的函数
            subagent = await self.agent_factory.create_subagent(
                prompt=full_prompt,
                model=self.model,
                name=f"hook_{event}",
                run_in_background=False  # 同步等待
            )
        
            # 4. 等待SubAgent完成
            # SubAgent会自动运行，完成后返回结果
            result = await subagent.wait_for_completion()
        
            # 5. 提取SubAgent的最后一条消息
            last_message = result.messages[-1]
        
            # 6. 尝试从消息中提取JSON
            message_text = last_message.content
        
            # 查找JSON块
            if "```json" in message_text:
                json_start = message_text.find("```json") + 7
                json_end = message_text.find("```", json_start)
                json_text = message_text[json_start:json_end].strip()
            elif "{" in message_text and "}" in message_text:
                # 提取第一个完整的JSON对象
                json_start = message_text.find("{")
                json_end = message_text.rfind("}") + 1
                json_text = message_text[json_start:json_end]
            else:
                # 没有找到JSON，返回默认
                return {
                    "allow": False,
                    "reason": "SubAgent未返回有效的决策",
                    "subagent_output": message_text
                }
        
            # 7. 解析JSON
            decision = json.loads(json_text)
        
            # 8. 验证必需字段
            if "allow" not in decision:
                decision["allow"] = False
                decision["reason"] = "SubAgent未给出明确决策"
        
            return decision
    
        except Exception as e:
            # 异常处理
            return {
                "allow": False,
                "reason": f"SubAgent执行失败：{str(e)}"
            }

### 🔍 Q1: "GitWorktree隔离机制"是如何实现的？

#### 源码位置
**文件**: `mewcode/worktree/manager.py`  
**类**: `WorktreeManager`

#### 完整源码（逐行注释）

```python
import os
import asyncio
import subprocess
from pathlib import Path
from typing import Optional, Dict, List
from dataclasses import dataclass

from mewcode.utils.git import GitClient


@dataclass
class WorktreeInfo:
    """
    工作树信息
  
    记录一个Git worktree的元数据
    """
    # 工作树的路径（绝对路径）
    path: str
  
    # 工作树使用的分支名称
    branch: str
  
    # 所属的Agent ID
    agent_id: str
  
    # 创建时间戳
    created_at: float
  
    # 是否活跃（Agent是否还在运行）
    is_active: bool = True


class WorktreeManager:
    """
    Git工作树管理器
  
    职责：
    1. 为每个Agent创建独立的git worktree
    2. 管理工作树的生命周期
    3. 清理不再使用的工作树
    4. 协调多个工作树之间的操作
  
    Git Worktree机制：
    - 允许同一个仓库有多个工作目录
    - 每个工作目录可以在不同的分支上
    - 各工作目录之间文件隔离，互不影响
    - 共享.git数据库，节省磁盘空间
    """
  
    def __init__(self, repo_path: str):
        """
        初始化工作树管理器
    
        参数:
            repo_path: Git仓库的根目录（包含.git的目录）
        """
        # 仓库根目录
        self.repo_path = Path(repo_path).resolve()
    
        # Git客户端（用于执行git命令）
        self.git = GitClient(repo_path)
    
        # 工作树存储目录（在仓库根目录的.git/worktrees/下）
        self.worktrees_dir = self.repo_path / ".git" / "mewcode_worktrees"
    
        # 确保目录存在
        self.worktrees_dir.mkdir(parents=True, exist_ok=True)
    
        # 工作树注册表：agent_id -> WorktreeInfo
        self.worktrees: Dict[str, WorktreeInfo] = {}
    
        # 锁（用于并发控制）
        self._lock = asyncio.Lock()
  
  
    async def create_worktree_for_agent(
        self,
        agent_id: str,
        branch_name: Optional[str] = None
    ) -> WorktreeInfo:
        """
        为Agent创建独立的工作树
    
        这是核心方法，实现Agent隔离
    
        流程：
        1. 生成唯一的工作树路径
        2. 创建新分支（基于当前分支）
        3. 创建git worktree
        4. 注册到管理器
    
        参数:
            agent_id: Agent的唯一标识
            branch_name: 可选的分支名称（默认自动生成）
    
        返回:
            WorktreeInfo对象，包含工作树的路径和分支信息
        """
        # 使用锁确保并发安全
        async with self._lock:
            # 1. 检查是否已存在
            if agent_id in self.worktrees:
                # 如果已经有工作树，直接返回
                return self.worktrees[agent_id]
        
            # 2. 生成分支名称
            if branch_name is None:
                # 自动生成：mewcode/agent_{agent_id}_{timestamp}
                # 例如：mewcode/agent_abc123_20240115103045
                import time
                timestamp = int(time.time())
                branch_name = f"mewcode/agent_{agent_id}_{timestamp}"
        
            # 3. 生成工作树路径
            # 路径格式：.git/mewcode_worktrees/agent_{agent_id}
            worktree_path = self.worktrees_dir / f"agent_{agent_id}"
        
            # 4. 确保路径不存在（清理旧的）
            if worktree_path.exists():
                # 如果存在旧的工作树，先删除
                await self._remove_worktree(str(worktree_path))
        
            # 5. 获取当前分支（作为新分支的基础）
            current_branch = await self.git.get_current_branch()
            # 例如：当前在main分支
        
            # 6. 创建git worktree
            # 命令：git worktree add -b {branch_name} {path} {base_branch}
            # 效果：
            #   - 创建新分支 branch_name，基于 current_branch
            #   - 在 worktree_path 创建新的工作目录
            #   - 将新工作目录切换到 branch_name 分支
            await self._execute_git_command([
                "worktree", "add",
                "-b", branch_name,        # 创建新分支
                str(worktree_path),       # 工作树路径
                current_branch            # 基于当前分支
            ])
        
            # 7. 创建工作树信息对象
            import time
            worktree_info = WorktreeInfo(
                path=str(worktree_path),
                branch=branch_name,
                agent_id=agent_id,
                created_at=time.time(),
                is_active=True
            )
        
            # 8. 注册到管理器
            self.worktrees[agent_id] = worktree_info
        
            # 9. 返回工作树信息
            return worktree_info
  
  
    async def remove_worktree_for_agent(self, agent_id: str) -> bool:
        """
        删除Agent的工作树
    
        清理流程：
        1. 从注册表移除
        2. 删除git worktree
        3. 删除分支（可选）
    
        参数:
            agent_id: Agent ID
    
        返回:
            True表示成功删除，False表示不存在
        """
        async with self._lock:
            # 1. 检查是否存在
            if agent_id not in self.worktrees:
                return False
        
            # 2. 获取工作树信息
            worktree_info = self.worktrees[agent_id]
        
            # 3. 从注册表移除
            del self.worktrees[agent_id]
        
            # 4. 删除git worktree
            await self._remove_worktree(worktree_info.path)
        
            # 5. 删除分支（可选，避免分支堆积）
            try:
                # 命令：git branch -D {branch_name}
                # -D表示强制删除（即使有未合并的提交）
                await self._execute_git_command([
                    "branch", "-D", worktree_info.branch
                ])
            except Exception as e:
                # 删除分支失败不影响工作树清理
                pass
        
            return True
  
  
    async def _remove_worktree(self, path: str):
        """
        删除git worktree（底层实现）
    
        注意：直接删除目录是不够的，必须使用git worktree remove
    
        参数:
            path: 工作树路径
        """
        try:
            # 命令：git worktree remove {path} --force
            # --force：即使有未提交的修改也删除
            await self._execute_git_command([
                "worktree", "remove",
                path,
                "--force"  # 强制删除
            ])
        except Exception as e:
            # 如果git命令失败，尝试手动删除目录
            import shutil
            if Path(path).exists():
                shutil.rmtree(path)
  
  
    async def get_worktree_for_agent(self, agent_id: str) -> Optional[WorktreeInfo]:
        """
        获取Agent的工作树信息
    
        参数:
            agent_id: Agent ID
    
        返回:
            WorktreeInfo对象，如果不存在则返回None
        """
        return self.worktrees.get(agent_id)
  
  
    async def list_all_worktrees(self) -> List[WorktreeInfo]:
        """
        列出所有工作树
    
        返回:
            所有工作树的列表
        """
        return list(self.worktrees.values())
  
  
    async def cleanup_inactive_worktrees(self):
        """
        清理不活跃的工作树
    
        定期调用此方法，清理已完成Agent的工作树
    
        判断依据：
        - Agent已经结束运行
        - 工作树超过一定时间未使用
        """
        async with self._lock:
            # 获取所有不活跃的工作树
            inactive = [
                agent_id
                for agent_id, info in self.worktrees.items()
                if not info.is_active
            ]
        
            # 删除不活跃的工作树
            for agent_id in inactive:
                await self.remove_worktree_for_agent(agent_id)
  
  
    async def mark_worktree_inactive(self, agent_id: str):
        """
        标记工作树为不活跃
    
        当Agent完成任务后调用
    
        参数:
            agent_id: Agent ID
        """
        if agent_id in self.worktrees:
            self.worktrees[agent_id].is_active = False
  
  
    async def _execute_git_command(self, args: List[str]) -> str:
        """
        执行git命令（底层实现）
    
        参数:
            args: git命令参数（不包含"git"本身）
    
        返回:
            命令输出
        """
        # 完整命令：git {args}
        cmd = ["git"] + args
    
        # 异步执行
        process = await asyncio.create_subprocess_exec(
            *cmd,
            cwd=str(self.repo_path),  # 在仓库根目录执行
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
    
        # 等待完成
        stdout, stderr = await process.communicate()
    
        # 检查退出码
        if process.returncode != 0:
            # 命令失败
            error_msg = stderr.decode("utf-8", errors="ignore")
            raise Exception(f"Git command failed: {' '.join(cmd)}\n{error_msg}")
    
        # 返回输出
        return stdout.decode("utf-8", errors="ignore")

```python
import asyncio
from typing import Dict, List, Optional
from dataclasses import dataclass
from datetime import datetime
import json


@dataclass
class Message:
    """
    Agent之间的消息
    """
    # 消息ID（唯一标识）
    id: str
  
    # 发送者Agent ID
    from_agent: str
  
    # 接收者Agent ID
    to_agent: str
  
    # 消息内容
    content: str
  
    # 消息类型（task/result/question/notification）
    message_type: str
  
    # 创建时间
    created_at: datetime
  
    # 元数据（可选的附加信息）
    metadata: Optional[Dict] = None


class MessageQueue:
    """
    Agent消息队列
  
    职责：
    1. 管理Agent之间的异步消息传递
    2. 确保消息不丢失
    3. 支持消息优先级
    4. 提供消息历史查询
  
    特点：
    - 异步非阻塞
    - 支持广播
    - 支持点对点
    - 支持消息过滤
    """
  
    def __init__(self):
        """
        初始化消息队列
        """
        # 每个Agent的消息队列：agent_id -> asyncio.Queue
        # asyncio.Queue是线程安全的异步队列
        self._queues: Dict[str, asyncio.Queue] = {}
    
        # 消息历史：存储所有发送过的消息
        # 用于调试和审计
        self._message_history: List[Message] = []
    
        # 锁（用于并发控制）
        self._lock = asyncio.Lock()
    
        # 消息ID计数器（用于生成唯一ID）
        self._message_counter = 0
  
  
    async def register_agent(self, agent_id: str):
        """
        注册Agent到消息队列
    
        每个Agent必须先注册才能收发消息
    
        参数:
            agent_id: Agent的唯一标识
        """
        async with self._lock:
            # 检查是否已注册
            if agent_id not in self._queues:
                # 为该Agent创建一个队列
                # maxsize=0表示无限大小（不限制消息数量）
                self._queues[agent_id] = asyncio.Queue(maxsize=0)
  
  
    async def unregister_agent(self, agent_id: str):
        """
        注销Agent
    
        Agent完成任务后调用，清理资源
    
        参数:
            agent_id: Agent ID
        """
        async with self._lock:
            # 检查是否已注册
            if agent_id in self._queues:
                # 删除队列
                del self._queues[agent_id]
  
  
    async def send_message(
        self,
        from_agent: str,
        to_agent: str,
        content: str,
        message_type: str = "task",
        metadata: Optional[Dict] = None
    ) -> str:
        """
        发送消息
    
        这是核心方法，实现异步消息传递
    
        参数:
            from_agent: 发送者Agent ID
            to_agent: 接收者Agent ID
            content: 消息内容
            message_type: 消息类型
            metadata: 附加元数据
    
        返回:
            消息ID
        """
        async with self._lock:
            # 1. 生成消息ID
            self._message_counter += 1
            message_id = f"msg_{self._message_counter:06d}"
        
            # 2. 创建消息对象
            message = Message(
                id=message_id,
                from_agent=from_agent,
                to_agent=to_agent,
                content=content,
                message_type=message_type,
                created_at=datetime.now(),
                metadata=metadata
            )
        
            # 3. 添加到历史记录
            self._message_history.append(message)
        
            # 4. 检查接收者是否已注册
            if to_agent not in self._queues:
                # 接收者未注册，消息会丢失
                raise ValueError(f"Agent {to_agent} not registered")
        
            # 5. 将消息放入接收者的队列
            # put_nowait()是非阻塞的，立即返回
            # 不会等待接收者读取消息
            await self._queues[to_agent].put(message)
        
            # 6. 返回消息ID
            return message_id
  
  
    async def receive_message(
        self,
        agent_id: str,
        timeout: Optional[float] = None
    ) -> Optional[Message]:
        """
        接收消息
    
        这是异步阻塞方法，会等待消息到来
    
        参数:
            agent_id: 接收者Agent ID
            timeout: 超时时间（秒），None表示永久等待
    
        返回:
            Message对象，如果超时则返回None
        """
        # 1. 检查Agent是否已注册
        if agent_id not in self._queues:
            raise ValueError(f"Agent {agent_id} not registered")
    
        # 2. 获取该Agent的队列
        queue = self._queues[agent_id]
    
        try:
            # 3. 从队列取出消息（异步等待）
            if timeout is None:
                # 永久等待，直到有消息
                message = await queue.get()
            else:
                # 带超时的等待
                message = await asyncio.wait_for(
                    queue.get(),
                    timeout=timeout
                )
        
            # 4. 返回消息
            return message
    
        except asyncio.TimeoutError:
            # 超时，返回None
            return None
  
  
    async def broadcast_message(
        self,
        from_agent: str,
        content: str,
        message_type: str = "notification",
        metadata: Optional[Dict] = None,
        exclude_agents: Optional[List[str]] = None
    ) -> List[str]:
        """
        广播消息
    
        向所有注册的Agent发送消息
    
        参数:
            from_agent: 发送者Agent ID
            content: 消息内容
            message_type: 消息类型
            metadata: 附加元数据
            exclude_agents: 排除的Agent列表（不发送给这些Agent）
    
        返回:
            所有发送的消息ID列表
        """
        # 默认排除列表
        exclude_agents = exclude_agents or []
    
        # 消息ID列表
        message_ids = []
    
        # 遍历所有已注册的Agent
        async with self._lock:
            for agent_id in self._queues.keys():
                # 跳过发送者自己
                if agent_id == from_agent:
                    continue
            
                # 跳过排除列表中的Agent
                if agent_id in exclude_agents:
                    continue
            
                # 发送消息
                message_id = await self.send_message(
                    from_agent=from_agent,
                    to_agent=agent_id,
                    content=content,
                    message_type=message_type,
                    metadata=metadata
                )
            
                message_ids.append(message_id)
    
        return message_ids
  
  
    async def check_pending_messages(self, agent_id: str) -> int:
        """
        检查待处理消息数量
    
        非阻塞方法，立即返回
    
        参数:
            agent_id: Agent ID
    
        返回:
            待处理消息的数量
        """
        if agent_id not in self._queues:
            return 0
    
        # qsize()返回队列中的消息数量
        return self._queues[agent_id].qsize()
  
  
    async def get_message_history(
        self,
        agent_id: Optional[str] = None,
        message_type: Optional[str] = None,
        limit: int = 100
    ) -> List[Message]:
        """
        获取消息历史
    
        用于调试和审计
    
        参数:
            agent_id: 如果指定，只返回该Agent相关的消息
            message_type: 如果指定，只返回该类型的消息
            limit: 返回的最大消息数量
    
        返回:
            消息列表（按时间倒序）
        """
        # 过滤消息
        filtered = self._message_history
    
        # 按Agent ID过滤
        if agent_id is not None:
            filtered = [
                msg for msg in filtered
                if msg.from_agent == agent_id or msg.to_agent == agent_id
            ]
    
        # 按消息类型过滤
        if message_type is not None:
            filtered = [
                msg for msg in filtered
                if msg.message_type == message_type
            ]
    
        # 倒序（最新的在前）
        filtered = list(reversed(filtered))
    
        # 限制数量
        return filtered[:limit]

# === 并行效率系数公式详解 ===

def calculate_efficiency_coefficient(
    sequential_time: float,
    parallel_time: float,
    num_agents: int
) -> float:
    """
    计算并行效率系数
  
    公式：
    η = T_ideal / T_actual
  
    其中：
    - T_ideal = T_sequential / N（理想并行时间）
    - T_actual = 实际并行时间
    - N = Agent数量
  
    理想值：
    - η = 1.0：完美并行（无任何开销）
    - η = 0.5：实际耗时是理想的2倍
    - η = 0.8：实际耗时是理想的1.25倍
  
    参数:
        sequential_time: 顺序执行时间
        parallel_time: 并行执行时间
        num_agents: Agent数量
  
    返回:
        效率系数（0-1之间）
    """
    # 1. 计算理想并行时间
    # 假设：任务均匀分配，无任何开销
    # 则：理想时间 = 总时间 / Agent数量
    ideal_time = sequential_time / num_agents
  
    # 2. 计算效率系数
    # η = 理想时间 / 实际时间
    efficiency = ideal_time / parallel_time
  
    # 3. 效率系数应该 ≤ 1.0
    # 如果 > 1.0，说明并行比理想还快（通常不可能，除非有缓存效应）
    if efficiency > 1.0:
        efficiency = 1.0
  
    return efficiency

from mewcode.hooks import HookManager, HookEvent
from mewcode.permissions.settings import PermissionSettings
from mewcode.permissions.rules import PermissionRule


class PermissionDecision(Enum):
    """
    权限决策结果
    """
    ALLOW = "allow"          # 允许
    DENY = "deny"            # 拒绝
    ASK_USER = "ask_user"    # 询问用户
    CONTINUE = "continue"    # 继续下一层判断


@dataclass
class PermissionRequest:
    """
    权限请求
  
    封装需要鉴权的操作信息
    """
    # 工具名称（如"Read", "Write", "Bash"）
    tool_name: str
  
    # 工具输入参数
    tool_input: Dict[str, Any]
  
    # Agent ID（发起请求的Agent）
    agent_id: str
  
    # 操作类型（read/write/execute）
    operation_type: str
  
    # 元数据（附加信息）
    metadata: Optional[Dict] = None


@dataclass
class PermissionResult:
    """
    权限判断结果
    """
    # 决策：允许/拒绝/询问用户
    decision: PermissionDecision
  
    # 决策来源（哪一层做的决策）
    source: str
  
    # 原因（为什么做这个决策）
    reason: str
  
    # 是否最终决策（True表示无需继续判断）
    is_final: bool = True


class PermissionInterceptor:
    """
    权限拦截器
  
    实现五层拦截模型：
    1. Prompt层：LLM通过工具描述self-limit
    2. Hook层：用户自定义Hook拦截
    3. Setting层：全局权限设置
    4. Rule层：细粒度规则匹配
    5. Manual层：人工确认
  
    每层都可以做出以下决策：
    - ALLOW：直接允许，跳过后续层
    - DENY：直接拒绝，跳过后续层
    - CONTINUE：无法决策，交给下一层
    - ASK_USER：需要人工确认
    """
  
    def __init__(
        self,
        hook_manager: HookManager,
        settings: PermissionSettings,
        rules: List[PermissionRule]
    ):
        """
        初始化拦截器
      
        参数:
            hook_manager: Hook管理器
            settings: 权限设置
            rules: 权限规则列表
        """
        # Hook管理器（用于Hook层拦截）
        self.hook_manager = hook_manager
      
        # 权限设置（用于Setting层）
        self.settings = settings
      
        # 权限规则（用于Rule层）
        self.rules = rules
  
  
    async def check_permission(
        self,
        request: PermissionRequest
    ) -> PermissionResult:
        """
        检查权限（五层拦截）
      
        流程：
        1. Prompt层（工具描述中的限制）
        2. Hook层（用户自定义拦截）
        3. Setting层（全局设置）
        4. Rule层（规则匹配）
        5. Manual层（人工确认）
      
        参数:
            request: 权限请求
      
        返回:
            权限判断结果
        """
        # === 第1层：Prompt层 ===
        # 说明：这一层在LLM调用工具之前就已经生效
        # LLM看到工具描述中的限制，会自我约束
        # 例如：工具描述中写"不要读取.env文件"
        # 这里只是记录，实际拦截在LLM决策阶段
      
        # 跳过Prompt层（因为已经在LLM侧生效）
      
        # === 第2层：Hook层 ===
        hook_result = await self._check_hook_layer(request)
        if hook_result.is_final:
            # Hook层做出了最终决策
            return hook_result
      
        # === 第3层：Setting层 ===
        setting_result = self._check_setting_layer(request)
        if setting_result.is_final:
            # Setting层做出了最终决策
            return setting_result
      
        # === 第4层：Rule层 ===
        rule_result = self._check_rule_layer(request)
        if rule_result.is_final:
            # Rule层做出了最终决策
            return rule_result
      
        # === 第5层：Manual层 ===
        # 如果前4层都没有做出最终决策，则需要人工确认
        return PermissionResult(
            decision=PermissionDecision.ASK_USER,
            source="Manual",
            reason="No automatic decision, manual confirmation required",
            is_final=True
        )
  
  
    async def _check_hook_layer(
        self,
        request: PermissionRequest
    ) -> PermissionResult:
        """
        Hook层检查
      
        调用用户自定义的Hook进行拦截
      
        参数:
            request: 权限请求
      
        返回:
            权限结果
        """
        # 1. 触发tool_call_before Hook
        # Hook可以返回 {"allow": true/false} 来决策
        hook_event = HookEvent.TOOL_CALL_BEFORE
      
        hook_payload = {
            "tool_name": request.tool_name,
            "tool_input": request.tool_input,
            "agent_id": request.agent_id
        }
      
        # 2. 执行Hook
        hook_responses = await self.hook_manager.trigger_event(
            event=hook_event,
            payload=hook_payload
        )
      
        # 3. 检查Hook响应
        for response in hook_responses:
            # 如果Hook明确返回allow字段
            if "allow" in response:
                if response["allow"] is True:
                    # Hook允许
                    return PermissionResult(
                        decision=PermissionDecision.ALLOW,
                        source="Hook",
                        reason=response.get("reason", "Allowed by hook"),
                        is_final=True
                    )
                elif response["allow"] is False:
                    # Hook拒绝
                    return PermissionResult(
                        decision=PermissionDecision.DENY,
                        source="Hook",
                        reason=response.get("reason", "Denied by hook"),
                        is_final=True
                    )
      
        # 4. Hook没有做决策，继续下一层
        return PermissionResult(
            decision=PermissionDecision.CONTINUE,
            source="Hook",
            reason="No hook decision",
            is_final=False
        )
  
  
    def _check_setting_layer(
        self,
        request: PermissionRequest
    ) -> PermissionResult:
        """
        Setting层检查
      
        根据全局权限设置判断
      
        参数:
            request: 权限请求
      
        返回:
            权限结果
        """
        # 1. 检查工具是否在黑名单中
        if request.tool_name in self.settings.blocked_tools:
            # 黑名单工具，直接拒绝
            return PermissionResult(
                decision=PermissionDecision.DENY,
                source="Setting",
                reason=f"Tool {request.tool_name} is in blocked list",
                is_final=True
            )
      
        # 2. 检查工具是否在白名单中
        if self.settings.allowed_tools:
            # 如果设置了白名单（非空列表）
            if request.tool_name in self.settings.allowed_tools:
                # 在白名单中，允许
                return PermissionResult(
                    decision=PermissionDecision.ALLOW,
                    source="Setting",
                    reason=f"Tool {request.tool_name} is in allowed list",
                    is_final=True
                )
            else:
                # 不在白名单中，拒绝
                return PermissionResult(
                    decision=PermissionDecision.DENY,
                    source="Setting",
                    reason=f"Tool {request.tool_name} not in allowed list",
                    is_final=True
                )
      
        # 3. 检查操作类型权限
        # 例如：settings.allow_read = true, allow_write = false
        if request.operation_type == "read":
            if not self.settings.allow_read:
                return PermissionResult(
                    decision=PermissionDecision.DENY,
                    source="Setting",
                    reason="Read operations are disabled",
                    is_final=True
                )
        elif request.operation_type == "write":
            if not self.settings.allow_write:
                return PermissionResult(
                    decision=PermissionDecision.DENY,
                    source="Setting",
                    reason="Write operations are disabled",
                    is_final=True
                )
        elif request.operation_type == "execute":
            if not self.settings.allow_execute:
                return PermissionResult(
                    decision=PermissionDecision.DENY,
                    source="Setting",
                    reason="Execute operations are disabled",
                    is_final=True
                )
      
        # 4. Setting层无明确决策，继续下一层
        return PermissionResult(
            decision=PermissionDecision.CONTINUE,
            source="Setting",
            reason="No setting restriction",
            is_final=False
        )
  
  
    def _check_rule_layer(
        self,
        request: PermissionRequest
    ) -> PermissionResult:
        """
        Rule层检查
      
        根据细粒度规则匹配判断
      
        参数:
            request: 权限请求
      
        返回:
            权限结果
        """
        # 1. 遍历所有规则
        for rule in self.rules:
            # 2. 检查规则是否匹配当前请求
            if rule.matches(request):
                # 3. 规则匹配，应用规则的决策
                if rule.action == "allow":
                    return PermissionResult(
                        decision=PermissionDecision.ALLOW,
                        source="Rule",
                        reason=f"Matched rule: {rule.name}",
                        is_final=True
                    )
                elif rule.action == "deny":
                    return PermissionResult(
                        decision=PermissionDecision.DENY,
                        source="Rule",
                        reason=f"Matched rule: {rule.name}",
                        is_final=True
                    )
                elif rule.action == "ask":
                    return PermissionResult(
                        decision=PermissionDecision.ASK_USER,
                        source="Rule",
                        reason=f"Matched rule: {rule.name} (requires confirmation)",
                        is_final=True
                    )
      
        # 4. 没有规则匹配，继续下一层
        return PermissionResult(
            decision=PermissionDecision.CONTINUE,
            source="Rule",
            reason="No rule matched",
            is_final=False
        )


# === 权限规则定义 ===

class PermissionRule:
    """
    权限规则
  
    用于Rule层的细粒度权限控制
    """
  
    def __init__(
        self,
        name: str,
        condition: str,
        action: str
    ):
        """
        初始化规则
      
        参数:
            name: 规则名称
            condition: 匹配条件（Python表达式）
            action: 动作（allow/deny/ask）
        """
        self.name = name
        self.condition = condition
        self.action = action
  
  
    def matches(self, request: PermissionRequest) -> bool:
        """
        检查规则是否匹配请求
      
        参数:
            request: 权限请求
      
        返回:
            True表示匹配
        """
        # 构建评估上下文
        context = {
            "tool_name": request.tool_name,
            "tool_input": request.tool_input,
            "agent_id": request.agent_id,
            "operation_type": request.operation_type,
        }
      
        try:
            # 评估条件表达式
            # 例如：condition = "tool_name == 'Read'"
            # eval()会返回True或False
            result = eval(self.condition, {"__builtins__": {}}, context)
            return bool(result)
        except Exception:
            # 条件表达式错误，不匹配
            return False
```

#### 源码位置

**类**: `MemoryExtractor`

#### 完整源码（逐行注释）

```python
import asyncio
from typing import List, Optional, Dict, Any
from dataclasses import dataclass
from datetime import datetime

from mewcode.llm import LLMClient
from mewcode.conversation import Message


@dataclass
class MemoryCandidate:
    """
    待提取的记忆候选
    """
    # 消息ID列表（相关的消息）
    message_ids: List[str]
  
    # 记忆类型（user/feedback/project/reference）
    memory_type: str
  
    # 记忆内容（由LLM提取）
    content: str
  
    # 提取时间
    extracted_at: datetime
  
    # 置信度（0-1）
    confidence: float


class MemoryExtractor:
    """
    异步内存提取器
  
    职责：
    1. 监听对话消息
    2. 在后台异步分析消息
    3. 提取有价值的记忆
    4. 写入Markdown文件
  
    关键特性：
    - 异步非阻塞（不影响主对话）
    - 批量处理（积累多条消息后统一分析）
    - 智能过滤（只提取有价值的信息）
    """
  
    def __init__(
        self,
        llm_client: LLMClient,
        memory_dir: str,
        batch_size: int = 10,
        extract_interval: float = 30.0
    ):
        """
        初始化提取器
      
        参数:
            llm_client: LLM客户端（用于分析）
            memory_dir: 记忆文件存储目录
            batch_size: 批量大小（积累多少条消息后触发提取）
            extract_interval: 提取间隔（秒）
        """
        # LLM客户端
        self.llm_client = llm_client
      
        # 记忆存储目录
        self.memory_dir = Path(memory_dir)
        self.memory_dir.mkdir(parents=True, exist_ok=True)
      
        # 批量大小
        self.batch_size = batch_size
      
        # 提取间隔
        self.extract_interval = extract_interval
      
        # 待处理的消息队列
        # 主对话的消息会被放入这个队列，等待后台处理
        self._pending_messages: asyncio.Queue = asyncio.Queue()
      
        # 后台任务
        self._extractor_task: Optional[asyncio.Task] = None
      
        # 是否运行中
        self._running = False
  
  
    async def start(self):
        """
        启动异步提取器
      
        在后台启动一个长期运行的任务
        """
        if self._running:
            return
      
        self._running = True
      
        # 启动后台提取任务
        self._extractor_task = asyncio.create_task(
            self._extraction_loop()
        )
      
        log.info("Memory extractor started")
  
  
    async def stop(self):
        """
        停止异步提取器
        """
        if not self._running:
            return
      
        self._running = False
      
        # 取消后台任务
        if self._extractor_task:
            self._extractor_task.cancel()
            try:
                await self._extractor_task
            except asyncio.CancelledError:
                pass
      
        log.info("Memory extractor stopped")
  
  
    async def add_message(self, message: Message):
        """
        添加消息到提取队列
      
        这是非阻塞的，立即返回
        主对话调用这个方法添加消息，不会等待提取完成
      
        参数:
            message: 对话消息
        """
        # 将消息放入队列（非阻塞）
        await self._pending_messages.put(message)
  
  
    async def _extraction_loop(self):
        """
        提取循环（后台任务）
      
        这是一个长期运行的异步任务
        持续从队列中取消息，批量提取记忆
        """
        # 消息缓冲区
        message_buffer: List[Message] = []
      
        # 上次提取时间
        last_extract_time = asyncio.get_event_loop().time()
      
        while self._running:
            try:
                # 1. 尝试从队列获取消息（带超时）
                try:
                    message = await asyncio.wait_for(
                        self._pending_messages.get(),
                        timeout=1.0  # 1秒超时
                    )
                  
                    # 添加到缓冲区
                    message_buffer.append(message)
              
                except asyncio.TimeoutError:
                    # 超时，没有新消息
                    pass
              
                # 2. 检查是否应该触发提取
                current_time = asyncio.get_event_loop().time()
                time_since_last_extract = current_time - last_extract_time
              
                should_extract = (
                    len(message_buffer) >= self.batch_size or  # 达到批量大小
                    (len(message_buffer) > 0 and time_since_last_extract >= self.extract_interval)  # 达到时间间隔
                )
              
                # 3. 如果应该提取
                if should_extract and message_buffer:
                    log.info(f"Extracting memories from {len(message_buffer)} messages")
                  
                    # 执行提取（异步）
                    await self._extract_memories(message_buffer)
                  
                    # 清空缓冲区
                    message_buffer.clear()
                  
                    # 更新时间
                    last_extract_time = current_time
          
            except Exception as e:
                log.error(f"Error in extraction loop: {e}")
                # 继续运行，不崩溃
  
  
    async def _extract_memories(self, messages: List[Message]):
        """
        从消息列表中提取记忆
      
        这是核心方法，调用LLM分析消息
      
        参数:
            messages: 消息列表
        """
        # 1. 构建提取提示词
        prompt = self._build_extraction_prompt(messages)
      
        # 2. 异步调用LLM
        # 使用Haiku快速模型，降低成本
        response = await self.llm_client.generate(
            prompt=prompt,
            model="claude-haiku-4.5",
            max_tokens=2000
        )
      
        # 3. 解析LLM响应
        candidates = self._parse_extraction_response(response.content[0].text)
      
        # 4. 过滤低置信度的候选
        # 只保留置信度 >= 0.7 的记忆
        filtered = [c for c in candidates if c.confidence >= 0.7]
      
        log.info(f"Extracted {len(filtered)} memories (from {len(candidates)} candidates)")
      
        # 5. 写入Markdown文件
        for candidate in filtered:
            await self._write_memory(candidate)
  
  
    def _build_extraction_prompt(self, messages: List[Message]) -> str:
        """
        构建记忆提取提示词
      
        参数:
            messages: 消息列表
      
        返回:
            提示词文本
        """
        # 1. 格式化消息内容
        messages_text = ""
        for msg in messages:
            role = msg.role.value
            content = msg.get_text_content()
            messages_text += f"\n[{role}]\n{content}\n"
      
        # 2. 构建提示词
        prompt = f"""
你是一个记忆提取专家。请分析以下对话消息，提取有价值的记忆。

记忆类型：
1. **user** - 用户的角色、偏好、知识背景
   示例："用户是高级Python开发者，熟悉FastAPI框架"
   
2. **feedback** - 用户对AI行为的反馈
   示例："不要使用pandas，改用polars，因为性能更好"
   
3. **project** - 项目的目标、进度、决策
   示例："项目使用PostgreSQL作为数据库，已完成用户认证模块"
   
4. **reference** - 外部资源的引用
   示例："API文档在 https://docs.example.com"

对话消息：
---
{messages_text}
---

请提取记忆，以JSON格式输出：
```json
[
  {{
    "type": "user" | "feedback" | "project" | "reference",
    "content": "记忆内容（1-2句话）",
    "confidence": 0.0-1.0,
    "reason": "为什么提取这条记忆"
  }}
]
```

要求：

1. 只提取**新的、有价值的**信息，不要重复已知的常识
2. 每条记忆要具体、可操作
3. 置信度评估要准确（不确定的设置为< 0.7）
4. 如果没有值得提取的记忆，返回空数组 []

请开始提取：
"""
        return prompt

    def _parse_extraction_response(self, response_text: str) -> List[MemoryCandidate]:
        """
        解析LLM的提取响应

        参数:
            response_text: LLM响应文本

        返回:
            记忆候选列表
        """
        import json
        import re

        # 1. 提取JSON块
        # 查找 ``json ... `` 或 ``...``
        json_match = re.search(r'``(?:json)?\s*(\[.*?\])\s*``', response_text, re.DOTALL)

        if not json_match:
            # 没有找到JSON块
            log.warning("No JSON found in extraction response")
            return []

        json_text = json_match.group(1)

        try:
            # 2. 解析JSON
            data = json.loads(json_text)

            # 3. 转换为MemoryCandidate对象
            candidates = []
            for item in data:
                candidate = MemoryCandidate(
                    message_ids=[],  # 稍后填充
                    memory_type=item["type"],
                    content=item["content"],
                    extracted_at=datetime.now(),
                    confidence=item.get("confidence", 0.8)
                )
                candidates.append(candidate)

            return candidates

        except json.JSONDecodeError as e:
            log.error(f"Failed to parse extraction JSON: {e}")
            return []

    async def _write_memory(self, candidate: MemoryCandidate):
        """
        将记忆写入Markdown文件

        参数:
            candidate: 记忆候选
        """
        # 1. 生成文件名
        # 格式：{type}_{timestamp}.md
        timestamp = int(candidate.extracted_at.timestamp())
        filename = f"{candidate.memory_type}_{timestamp}.md"

        file_path = self.memory_dir / filename

# 2. 生成Markdown内容
        content = f"""---
type:
created_at:
confidence:
-----------

{candidate.content}
"""

        # 3. 写入文件
        with open(file_path, "w", encoding="utf-8") as f:
            f.write(content)

        log.info(f"Memory written: {filename}")

```
