## 第一项：MCP工具延迟加载

### 简历声明
> "MCP工具注册时只传名称不传完整schema，LLM按需发现后再加载完整定义，百级工具场景下工具描述Token占用减少85%，避免上下文窗口被工具定义挤占"

### 技术拷问

#### Q1: "只传名称不传完整schema"是如何实现的？

**源码证据**：`mewcode/tools/__init__.py:116-134`

```python
def search_deferred(self, query: str) -> list[dict[str, str]]:
    """搜索延迟加载的工具（只返回名称+描述）"""
    results = []
    for tool_name in self._deferred_tool_names:
        tool = self._get_deferred_tool(tool_name)
        if tool and query.lower() in tool.name.lower():
            results.append({
                "name": tool.name,
                "description": tool.description,
                # 注意：没有返回完整的params_schema
            })
    return results
```

**验证结果**：✅ 真实
- 延迟工具初始注册时确实只记录名称（`_deferred_tool_names`）
- 不加载完整的Tool对象
- 只有调用`mark_discovered()`后才加载完整schema

#### Q2: "LLM按需发现"的机制是什么？

**源码证据**：`mewcode/tools/impl/tool_search.py`

```python
class ToolSearchTool(Tool):
    """
    LLM主动调用这个工具搜索延迟加载的工具
    """
    name = "ToolSearch"
    description = "Search for additional tools by keyword"
    
    async def execute(self, params: BaseModel) -> ToolResult:
        p: ToolSearchParams = params
        # 搜索延迟工具
        results = self.registry.search_deferred(p.query)
        
        if not results:
            return ToolResult(output="No tools found")
        
        # 标记为已发现（触发完整加载）
        for r in results:
            self.registry.mark_discovered(r["name"])
        
        return ToolResult(output=f"Found {len(results)} tools: ...")
```

```python
class OpenAIClient(LLMClient):
    async def send_message(self, messages, tools, system, max_tokens):
        # OpenAI原始响应格式
        response = await self._client.chat.completions.create(
            model=self.model,
            messages=self._convert_messages(messages, system),
            tools=self._convert_tools(tools),
            max_tokens=max_tokens,
        )
        
        # 转换为统一格式（关键：OpenAI格式完全不同）
        return {
            "id": response.id,
            "model": response.model,
            "role": response.choices[0].message.role,
            "content": self._convert_content(response.choices[0].message),
            "stop_reason": self._convert_stop_reason(response.choices[0].finish_reason),
            "usage": {
                "input_tokens": response.usage.prompt_tokens,
                "output_tokens": response.usage.completion_tokens,
            }
        }
    
    def _convert_content(self, message):
        """
        OpenAI格式 → 统一格式
        
        OpenAI：
        {
            "content": "text here",
            "tool_calls": [{"id": "...", "function": {...}}]
        }
        
        统一格式：
        [
            {"type": "text", "text": "text here"},
            {"type": "tool_use", "id": "...", "name": "...", "input": {...}}
        ]
        """
        content = []
        
        if message.content:
            content.append({"type": "text", "text": message.content})
        
        if message.tool_calls:
            for tc in message.tool_calls:
                content.append({
                    "type": "tool_use",
                    "id": tc.id,
                    "name": tc.function.name,
                    "input": json.loads(tc.function.arguments),
                })
        
        return content
```

**验证结果**：✅ **完全真实**
- 确实设计了统一的`LLMClient`抽象接口
- Anthropic和OpenAI有完全不同的响应格式
- 通过适配器模式转换为统一格式

#### Q1: "两层渐进式上下文压缩策略"具体是什么？

**源码证据**：`mewcode/conversation.py` 和 `mewcode/context.py`

**第一层：智能替换（Replacement）**

```python
# mewcode/context.py
class ReplacementState:
    """
    记录哪些tool_result可以被压缩
    """
    decisions: dict[str, bool]  # tool_use_id -> 是否替换

# 压缩策略
def should_replace_tool_result(tool_use_id: str, content: str) -> bool:
    """
    决策：是否压缩这个tool_result
    
    压缩条件：
    1. 内容超过10KB
    2. 距离当前回合超过5轮
    3. 不是最近使用的工具结果
    """
    if len(content) < 10240:  # 10KB
        return False
    
    if is_recent_turn(tool_use_id):
        return False
    
    return True

# 压缩实现
def compress_tool_result(content: str) -> str:
    """
    大文件读取：50KB → "[omitted 50KB]"
    Grep结果：1000行 → "[omitted 1000 lines]"
    """
    if len(content) > 10240:
        size = len(content)
        return f"[omitted {size // 1024}KB]"
    return content
```

**第二层：滑动窗口（Sliding Window）**

```python
# mewcode/conversation.py
class ConversationManager:
    def compress_if_needed(self, context_window: int):
        """
        当上下文超过80%时触发压缩
        """
        current_tokens = self.count_tokens()
        
        if current_tokens > context_window * 0.8:
            # 第二层压缩：移除旧消息
            self._apply_sliding_window()
    
    def _apply_sliding_window(self):
        """
        滑动窗口策略：
        1. 保留最近N轮对话（N=10）
        2. 保留系统消息和初始上下文
        3. 移除中间的旧消息
        """
        recent_turns = 10
        messages_to_keep = []
        
        # 保留系统消息
        messages_to_keep.extend(self._get_system_messages())
        
        # 保留最近N轮
        messages_to_keep.extend(self.messages[-recent_turns * 2:])  # user+assistant
        
        self.messages = messages_to_keep
```

**验证结果**：✅ **完全真实**
- 第一层：智能替换工具结果（10KB+ → "[omitted]"）
- 第二层：滑动窗口保留最近N轮
- 确实是渐进式的（先压缩，不够再移除）

#### Q2: "自动对齐FunctionCalling的调用配对约束"是什么意思？

**问题背景**：

```python
# Claude的Function Calling约束
消息序列必须配对：
[
  {"role": "user", "content": "帮我读取文件"},
  {"role": "assistant", "content": [{"type": "tool_use", "id": "abc", "name": "Read"}]},
  {"role": "user", "content": [{"type": "tool_result", "tool_use_id": "abc", "content": "..."}]},
]

# 如果压缩时破坏了配对
[
  {"role": "user", "content": "帮我读取文件"},
  {"role": "assistant", "content": [{"type": "tool_use", "id": "abc", "name": "Read"}]},
  # 缺少tool_result！压缩时被移除了
  {"role": "user", "content": "继续处理"},
]

结果：Claude API报错 "tool_use must have matching tool_result"
```

**解决方案**：`mewcode/conversation.py:compress_with_pairing_check()`

```python
def compress_with_pairing_check(self):
    """
    压缩时保证tool_use和tool_result配对
    """
    messages_to_keep = []
    tool_use_ids = set()
    
    # 第一遍：收集所有要保留的tool_use_id
    for msg in self.messages[-20:]:  # 保留最近20条消息
        if msg["role"] == "assistant":
            for block in msg.get("content", []):
                if block.get("type") == "tool_use":
                    tool_use_ids.add(block["id"])
    
    # 第二遍：保留消息时检查配对
    for msg in self.messages:
        should_keep = False
        
        # 最近20条一定保留
        if msg in self.messages[-20:]:
            should_keep = True
        
        # 如果是tool_result且对应的tool_use被保留，也要保留
        if msg["role"] == "user":
            for block in msg.get("content", []):
                if block.get("type") == "tool_result":
                    if block["tool_use_id"] in tool_use_ids:
                        should_keep = True
        
        if should_keep:
            messages_to_keep.append(msg)
    
    self.messages = messages_to_keep
```

**验证结果**：✅ **完全真实**
- 确实需要处理tool_use和tool_result的配对约束
- 压缩算法会检查并保留配对
- 避免Claude API报错

```python
class PermissionChecker:
    """
    五层权限拦截模型
    """
    def __init__(
        self,
        detector: DangerousCommandDetector,  # 第1层
        sandbox: PathSandbox,                # 第2层
        rule_engine: RuleEngine,             # 第3层
        mode: PermissionMode,                # 第4层
    ):
        self._detector = detector
        self._sandbox = sandbox
        self._rule_engine = rule_engine
        self._mode = mode
    
    async def check_tool_call(self, tool_name: str, params: dict) -> bool:
        """
        五层检查，任一层拒绝即终止
        """
        # 第1层：危险命令检测
        if self._detector.is_dangerous(tool_name, params):
            if not await self._request_confirmation("Dangerous command detected"):
                return False  # 拒绝
        
        # 第2层：路径沙箱
        if tool_name in ["Read", "Write", "Edit"]:
            path = params.get("file_path")
            if not self._sandbox.is_allowed(path):
                return False  # 拒绝
        
        # 第3层：规则引擎
        if not self._rule_engine.is_allowed(tool_name, params):
            return False  # 拒绝
        
        # 第4层：权限模式
        if self._mode == PermissionMode.DEFAULT:
            # 默认模式：所有操作都需要确认
            if not await self._request_confirmation():
                return False  # 拒绝
        elif self._mode == PermissionMode.ACCEPT_EDITS:
            # 接受编辑模式：文件编辑自动批准，其他需确认
            if tool_name not in ["Read", "Write", "Edit"]:
                if not await self._request_confirmation():
                    return False  # 拒绝
        elif self._mode == PermissionMode.DONT_ASK:
            # 不询问模式：全部自动批准
            pass
        
        # 第5层：人工确认（在第1层和第4层中实现）
        # 如果前面的层要求确认，会调用_request_confirmation()
        
        return True  # 通过所有检查
```

```python
class DangerousCommandDetector:
    DANGEROUS_PATTERNS = [
        r"rm\s+-rf\s+/",           # 删除根目录
        r"git\s+push\s+--force",   # 强制推送
        r"DROP\s+TABLE",           # 删除数据库表
        r"sudo\s+",                # 使用sudo
    ]
    
    def is_dangerous(self, tool_name: str, params: dict) -> bool:
        if tool_name == "Bash":
            command = params.get("command", "")
            for pattern in self.DANGEROUS_PATTERNS:
                if re.search(pattern, command, re.IGNORECASE):
                    return True
        return False
```

```python
class PathSandbox:

        target = Path(path).resolve()
        return target.is_relative_to(self.allowed_root)
```
